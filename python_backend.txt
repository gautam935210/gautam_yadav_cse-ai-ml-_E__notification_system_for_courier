# 3. API Endpoint to Update Parcel Status & Send WhatsApp Notifications
@app.route('/update-status', methods=['POST'])
def update_status():
    try:
        data = request.json
        tracking_id = data['tracking_id']
        new_status = data['status']
        
        conn = connect_db()
        cursor = conn.cursor()
        
        cursor.execute("UPDATE parcels SET status=%s WHERE tracking_id=%s", (new_status, tracking_id))
        conn.commit()
        
        cursor.execute("SELECT whatsapp_number FROM parcels WHERE tracking_id=%s", (tracking_id,))
        result = cursor.fetchone()
        
        if result:
            whatsapp = result[0]
            message = f"Your parcel ({tracking_id}) is now '{new_status}'."
            send_whatsapp(whatsapp, message)
            return jsonify({"message": "Status updated and WhatsApp notification sent!"})
        else:
            return jsonify({"error": "Tracking ID not found."}), 404
    except Exception as e:
        return jsonify({"error": str(e)}), 500
